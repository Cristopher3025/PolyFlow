# Gramática propuesta

REGLA ::= "REGLA" CONDICION "->" ACCION
CONDICION ::= CAMPO OPERADOR VALOR
CAMPO ::= "TEMPERATURA" | "PRECIPITACION" | "VIENTO" | "BATERIA"
OPERADOR ::= ">" | "<" | ">=" | "<=" | "=="
ACCION ::= "ALERTA_" IDENTIFICADOR

Ejemplos:
REGLA TEMPERATURA > 35 -> ALERTA_CALOR
REGLA VIENTO >= 60 -> ALERTA_VIENTO
